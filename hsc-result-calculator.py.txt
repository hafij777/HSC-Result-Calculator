
def get_valid_number(prompt, min_val, max_val):
    while True:
        try:
            value = int(input(prompt))
            if min_val <= value <= max_val:
                return value
            else:
                print(f"Please enter number between {min_val} and {max_val}")
        except:
            print("Invalid input! Only number allowed.")


def grade(mcq, cq, practical, subject, full_mcq, full_cq, full_practical):
    total = mcq + cq + practical
    full_total = full_mcq + full_cq + full_practical

    if mcq < (full_mcq * 0.33) or cq < (full_cq * 0.33):
        result = "Fail"
    elif total >= full_total * 0.90:
        result = "Golden A+"
    elif total >= full_total * 0.80:
        result = "A+"
    elif total >= full_total * 0.70:
        result = "A"
    elif total >= full_total * 0.60:
        result = "A-"
    elif total >= full_total * 0.50:
        result = "B"
    elif total >= full_total * 0.40:
        result = "C"
    elif total >= full_total * 0.33:
        result = "D"
    else:
        result = "Fail"

    print(f"\n{subject}")
    print(f"MCQ: {mcq}/{full_mcq} | CQ: {cq}/{full_cq} | Practical: {practical}/{full_practical}")
    print(f"Total: {total}/{full_total} → Grade: {result}")
    print("-" * 40)


print("===== Student Result System =====")

reg = get_valid_number("Registration Number (max 10 digits): ", 0, 9999999999)

if group == "SCIENCE":
    subjects = common_subjects + science_subjects
